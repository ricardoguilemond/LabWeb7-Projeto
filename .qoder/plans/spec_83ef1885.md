## Task 4 — Script `ALTER TABLE` para banco de homologação (execução manual)

```sql
BEGIN;

-- ============================================================
-- 1) Adicionar colunas como NULLABLE temporariamente
-- ============================================================
ALTER TABLE "Postos" ADD COLUMN "InstituicaoId" INT;
ALTER TABLE "Postos" ADD COLUMN "SiglaPosto" VARCHAR(20);

-- ============================================================
-- 2) Inferência ampliada: ExamesRealizados → ExamesRealizadosAM → Requisitar
--    Atribui ao Posto a Instituicao do PRIMEIRO exame que o referencia.
-- ============================================================
WITH inferencia AS (
    SELECT DISTINCT ON (sub."PostoId") sub."PostoId", sub."InstituicaoId"
    FROM (
        SELECT "PostoId", "InstituicaoId", "Id", 1 AS prioridade FROM "ExamesRealizados"   WHERE "PostoId" IS NOT NULL
        UNION ALL
        SELECT "PostoId", "InstituicaoId", "Id", 2 AS prioridade FROM "ExamesRealizadosAM" WHERE "PostoId" IS NOT NULL
        UNION ALL
        SELECT "PostoId", "InstituicaoId", "Id", 3 AS prioridade FROM "Requisitar"         WHERE "PostoId" IS NOT NULL
    ) sub
    ORDER BY sub."PostoId", sub.prioridade ASC, sub."Id" ASC
)
UPDATE "Postos" p
SET "InstituicaoId" = i."InstituicaoId"
FROM inferencia i
WHERE p."Id" = i."PostoId";

-- ============================================================
-- 3) Excluir Postos órfãos (sem exame referenciando — D1)
-- ============================================================
DELETE FROM "Postos" WHERE "InstituicaoId" IS NULL;

-- ============================================================
-- 4) Duplicar Postos usados por mais de uma Instituicao (D2)
--    Para cada combinação (PostoId original, InstituicaoId outra),
--    cria uma cópia nova e religa os exames daquelas Instituicoes.
-- ============================================================
DO $$
DECLARE
    r RECORD;
    novo_id INT;
BEGIN
    -- Loop pelas combinações que NÃO coincidem com a Instituicao já gravada no Posto original
    FOR r IN
        SELECT t."PostoId", t."InstituicaoId" AS inst_extra,
               p."NomePosto", p."Responsavel", p."Telefone", p."Endereco",
               p."Logradouro", p."Numero", p."Bairro", p."Complemento",
               p."Cidade", p."UF", p."CEP"
        FROM (
            SELECT "PostoId", "InstituicaoId" FROM "ExamesRealizados"   WHERE "PostoId" IS NOT NULL
            UNION
            SELECT "PostoId", "InstituicaoId" FROM "ExamesRealizadosAM" WHERE "PostoId" IS NOT NULL
            UNION
            SELECT "PostoId", "InstituicaoId" FROM "Requisitar"         WHERE "PostoId" IS NOT NULL
        ) t
        JOIN "Postos" p ON p."Id" = t."PostoId"
        WHERE t."InstituicaoId" <> p."InstituicaoId"
    LOOP
        -- Reaproveita o primeiro Id vago (gap-filling, padrão LabWeb7)
        SELECT COALESCE(MIN(seq."n"), (SELECT MAX("Id") FROM "Postos") + 1)
        INTO novo_id
        FROM generate_series(1, (SELECT COALESCE(MAX("Id"),0)+1 FROM "Postos")) AS seq("n")
        WHERE NOT EXISTS (SELECT 1 FROM "Postos" px WHERE px."Id" = seq."n");

        INSERT INTO "Postos" ("Id","NomePosto","Responsavel","Telefone","Endereco",
                              "Logradouro","Numero","Bairro","Complemento",
                              "Cidade","UF","CEP","InstituicaoId")
        VALUES (novo_id, r."NomePosto", r."Responsavel", r."Telefone", r."Endereco",
                r."Logradouro", r."Numero", r."Bairro", r."Complemento",
                r."Cidade", r."UF", r."CEP", r.inst_extra);

        -- Religar exames das 3 tabelas que tinham PostoId original + Instituicao "extra"
        UPDATE "ExamesRealizados"   SET "PostoId" = novo_id WHERE "PostoId" = r."PostoId" AND "InstituicaoId" = r.inst_extra;
        UPDATE "ExamesRealizadosAM" SET "PostoId" = novo_id WHERE "PostoId" = r."PostoId" AND "InstituicaoId" = r.inst_extra;
        UPDATE "Requisitar"         SET "PostoId" = novo_id WHERE "PostoId" = r."PostoId" AND "InstituicaoId" = r.inst_extra;
    END LOOP;
END $$;

-- ============================================================
-- 5) Preencher SiglaPosto para registros existentes (D6 — ver dúvidas)
-- ============================================================
UPDATE "Postos"
SET "SiglaPosto" = UPPER(SUBSTRING("NomePosto" FROM 1 FOR 20))
WHERE "SiglaPosto" IS NULL;

-- ============================================================
-- 6) Tornar colunas NOT NULL
-- ============================================================
ALTER TABLE "Postos" ALTER COLUMN "InstituicaoId" SET NOT NULL;
ALTER TABLE "Postos" ALTER COLUMN "SiglaPosto"    SET NOT NULL;

-- ============================================================
-- 7) Adicionar FK e índice de performance
-- ============================================================
ALTER TABLE "Postos"
    ADD CONSTRAINT "iPostos_Instituicao"
    FOREIGN KEY ("InstituicaoId") REFERENCES "Instituicao"("Id")
    ON DELETE RESTRICT;

CREATE INDEX IF NOT EXISTS "iPostos_InstituicaoId" ON "Postos" ("InstituicaoId");

COMMIT;
```

> O bloco está dentro de `BEGIN; ... COMMIT;` para que qualquer falha não deixe o banco em estado intermediário.
